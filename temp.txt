const { MongoClient } = require('mongodb');

async function migrateToCloud() {
  const localUri = 'mongodb://localhost:27017/leehomes';
  const cloudUri = process.env.MONGODB_ATLAS_URI;
  
  const localClient = new MongoClient(localUri);
  const cloudClient = new MongoClient(cloudUri);
  
  try {
    await localClient.connect();
    await cloudClient.connect();
    
    const localDb = localClient.db('leehomes');
    const cloudDb = cloudClient.db('leehomes');
    
    // Migrate collections
    const collections = ['employees', 'attendance', 'departments', 'settings'];
    
    for (const collName of collections) {
      console.log(`Migrating ${collName}...`);
      const docs = await localDb.collection(collName).find({}).toArray();
      
      if (docs.length > 0) {
        await cloudDb.collection(collName).insertMany(docs);
        console.log(`✅ Migrated ${docs.length} documents to ${collName}`);
      }
    }
    
  } finally {
    await localClient.close();
    await cloudClient.close();
  }
}

migrateToCloud().catch(console.error);

class AttendancePartitioner {
  static async partitionByMonth() {
    const db = getDatabase();
    
    // Tạo collection theo tháng
    const currentMonth = new Date().toISOString().slice(0, 7); // "2025-09"
    const collectionName = `attendance_${currentMonth.replace('-', '_')}`;

  }
  
  static async queryAttendanceRange(startDate, endDate, employeeId) {
    const months = this.getMonthsBetween(startDate, endDate);
    const results = [];
    
    for (const month of months) {
      const collection = `attendance_${month.replace('-', '_')}`;
      const monthData = await db.collection(collection)
        .find({
          employeeId,
          date: { $gte: startDate, $lte: endDate }
        }).toArray();
      
      results.push(...monthData);
    }
    
    return results;
  }
}

class DataLifecycleManager {
  static async setupArchiving() {
    // Archive data > 2 năm sang collection khác
    const archiveDate = new Date();
    archiveDate.setFullYear(archiveDate.getFullYear() - 2);
    
    const pipeline = [
      { $match: { date: { $lt: archiveDate } } },
      { $out: "attendance_archive" }
    ];
    
    await db.collection('attendance').aggregate(pipeline);
    
    // Xóa data cũ khỏi main collection
    await db.collection('attendance').deleteMany({
      date: { $lt: archiveDate }
    });
  }
  
  // Compression cho archived data
  static async compressArchive() {
    const archiveStats = await db.collection('attendance_archive').aggregate([
      {
        $group: {
          _id: {
            employeeId: "$employeeId",
            date: { $dateToString: { format: "%Y-%m-%d", date: "$date" } }
          },
          totalHours: { $sum: "$hours" },
          checkCount: { $sum: 1 },
          firstCheck: { $min: "$timestamp" },
          lastCheck: { $max: "$timestamp" }
        }
      },
      { $out: "daily_attendance_summary" }
    ]);
  }

import { Redis } from 'ioredis';

class AttendanceCache {
  static redis = new Redis(process.env.REDIS_URL);
  
  static async getCachedEmployeeAttendance(employeeId, date) {
    const key = `attendance:${employeeId}:${date}`;
    const cached = await this.redis.get(key);
    
    if (cached) return JSON.parse(cached);
    
    // Cache miss - fetch from DB
    const data = await db.collection('attendance')
      .find({ employeeId, date })
      .toArray();
    
    // Cache for 1 hour
    await this.redis.setex(key, 3600, JSON.stringify(data));
    return data;
  }
  
  static async invalidateEmployeeCache(employeeId) {
    const pattern = `attendance:${employeeId}:*`;
    const keys = await this.redis.keys(pattern);
    if (keys.length > 0) {
      await this.redis.del(...keys);
    }
  }
}
